# Exercice 4 — Le Deployment de votre site.
# Appliquer avec : kubectl apply -f k8s/site/
# Aide : kubectl explain deployment.spec.template.spec.containers
apiVersion: apps/v1
kind: Deployment
metadata:
  name: site
  labels:
    app: site
spec:
  replicas: 3                      # Exercice 5 : passé à 3 réplicas
  selector:
    matchLabels:
      app: site                    # correspond aux labels du template ci-dessous
  template:
    metadata:
      labels:
        app: site
    spec:
      containers:
        - name: nginx
          image: nginx:1.28        # version précise, pas "latest"
          ports:
            - containerPort: 80    # le port sur lequel nginx écoute
          readinessProbe:            # Bonus B : trafic envoyé seulement aux pods prêts
            httpGet:
              path: /
              port: 80
            initialDelaySeconds: 2
            periodSeconds: 5
          resources:                 # Bonus B : réservation et plafond
            requests:
              cpu: 50m
              memory: 32Mi
            limits:
              cpu: 100m
              memory: 64Mi
          volumeMounts:
            - name: html
              mountPath: /usr/share/nginx/html   # le dossier servi par nginx (cf. exercice 3)
      volumes:
        - name: html
          configMap:
            name: site-html        # le nom de la ConfigMap de configmap.yaml
